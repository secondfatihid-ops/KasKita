SET @staff = (SELECT id FROM users WHERE email = 'staff@kaskita.test');
SET @cat   = (SELECT id FROM categories WHERE name = 'Sales');

-- Each of these should FAIL:
INSERT INTO transactions (user_id, category_id, amount, transaction_date)
  VALUES (@staff, @cat, 0, CURDATE());                          -- 3819: amount must be > 0
INSERT INTO users (role_id, name, email, password_hash)
  VALUES (2, 'X', 'staff@kaskita.test', 'x');                   -- 1062: duplicate email
INSERT INTO users (role_id, name, email, password_hash)
  VALUES (99, 'X', 'x@test.com', 'x');                          -- 1452: role does not exist

-- This should WORK:
INSERT INTO transactions (user_id, category_id, amount, transaction_date, note)
  VALUES (@staff, @cat, 150000.00, CURDATE(), 'Test sale');
SET @tx = LAST_INSERT_ID();
INSERT INTO attachments (transaction_id, file_name, file_url, public_id, mime_type, file_size)
  VALUES (@tx, 'r.jpg', 'https://example.com/r.jpg', 'kaskita/r', 'image/jpeg', 100000);

-- Each of these should FAIL:
INSERT INTO attachments (transaction_id, file_name, file_url, public_id, mime_type, file_size)
  VALUES (@tx, 'r2.jpg', 'https://example.com/r2.jpg', 'kaskita/r2', 'image/jpeg', 100000);  -- 1062: one receipt per transaction
UPDATE attachments SET mime_type = 'application/pdf' WHERE transaction_id = @tx;              -- 3819: only JPG/PNG
UPDATE attachments SET file_size = 3000000 WHERE transaction_id = @tx;                        -- 3819: over 2 MB
DELETE FROM users WHERE id = @staff;                                                          -- 1451: user has transactions

-- Deleting the transaction should remove its receipt row too:
DELETE FROM transactions WHERE id = @tx;
SELECT COUNT(*) FROM attachments;                               -- expect 0
